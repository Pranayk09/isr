import re
import os
from collections import defaultdict

from whoosh.index import create_in
from whoosh.qparser import QueryParser
from whoosh.fields import Schema, TEXT, ID
documents = [
    {"id": "1", "content": "Machine learning is a subset of artificial intelligence."},
    {"id": "2", "content": "Deep learning uses neural networks."},
    {"id": "3", "content": "Python is widely used in data science."},
    {"id": "4", "content": "Artificial intelligence includes machine learning and deep learning."},
    {"id": "5", "content": "Data science uses Python and machine learning."}
]

# create inverted index

inverted_index = defaultdict(list)

for doc in documents:
    words = re.findall(r"\w+", doc['content'].lower())
    for word in set(words):
        inverted_index[word].append(doc['id'])
    
print("Inverted index:")

for word in sorted(inverted_index):
    print(f"{word:15} -> {inverted_index[word]}")




#Search

query = input("Enter query: ").lower().strip()

if query in inverted_index:
    print("found doc")
    for doc_id in inverted_index[query]:
        for doc in documents:
            if doc['id'] == doc_id:
                print("Documents ", doc['id'])
                print(doc['content'])
                print()
else:
    print("doc not found")




# whoosh index 

index_dir = "whoosh_index_p"

os.mkdir(index_dir)

schema = Schema(
    id=ID(stored=True),
    content=TEXT(stored=True)
)

ix = create_in(index_dir,schema)

writer = ix.writer()

for doc in documents:
    writer.add_document(
        id=doc['id'],
        content=doc['content']
    )

writer.commit()

print("whoosh_index created")


# boolean search


with ix.searcher() as searcher:
    parser = QueryParser("content", ix.schema)

    boolean_input = input("Enter query")

    boolean_query = parser.parse(boolean_input)

    results = searcher.search(boolean_query);

    if len(results) == 0:
        print("No doc found")
    else:
        for res in results:
            print("document ", res['id'])
            print(res['content'])
            print()
